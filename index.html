Tabla de Perfiles CREATE TABLE public.profiles ( id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE, email TEXT, role TEXT DEFAULT 'user', -- 'admin' o 'user' credits INT DEFAULT 20 ); -- 2.
